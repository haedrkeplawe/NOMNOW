التاكد من القيم التي بداخل السله قبل تحولها لاوردر التاكد من توفر المحتويات وفي حال تغيير السعر اخبار المستخدم بذلك


// update DE — عند إنشاء الطلب، نحسب الضريبة بدقة ونُرفق تفصيلها في الـ response للعرض في الفاتورة
exports.createOrder = async (req, res) => {
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const userId = req.user._id ?? req.user.id;

    if (req.user.status === "blocked") {
      await session.abortTransaction();
      session.endSession();
      return res.status(403).json({ message: "Your account is blocked" });
    }

    const { deliveryAddress, notes, paymentMethod: reqPaymentMethod, confirmed = false } = req.body;

    await Order.deleteMany({
      userId,
      orderStatus: "not_confirmed",
    }).session(session);

    const cart = await Cart.findOne({ userId }).session(session);

    if (!cart || cart.items.length === 0) {
      await session.abortTransaction();
      session.endSession();
      return res.status(400).json({ message: "Cart is empty" });
    }

    // ── جلب المطعم ────────────────────────────────────────────
    const restaurant = await Restaurant.findById(cart.restaurantId).select(
      "taxRate paymentMethods country status",
    );
    if (!restaurant) {
      await session.abortTransaction();
      session.endSession();
      return res.status(404).json({ message: "Restaurant not found" });
    }
    if (restaurant.status !== "open") {
      await session.abortTransaction();
      session.endSession();
      return res.status(400).json({ message: "Restaurant is currently closed" });
    }
    if (restaurant.country !== (req.user.country || "SY")) {
      await session.abortTransaction();
      session.endSession();
      return res.status(400).json({
        message: "You cannot order from a restaurant in a different country",
      });
    }

    // ── التحقق من السلة مقابل DB ──────────────────────────────
    const foodIds = cart.items.map((i) => i.foodId);
    const freshFoods = await Food.find({ _id: { $in: foodIds } }).select(
      "price sizes extras status name",
    );
    const foodMap = new Map(freshFoods.map((f) => [f._id.toString(), f]));

    const changes = []; // التغييرات التي سيُخبَر بها المستخدم

    for (const cartItem of cart.items) {
      const food = foodMap.get(cartItem.foodId.toString());

      // 1) الطعام محذوف أو غير متاح
      if (!food || food.status !== "available") {
        changes.push({
          type: "unavailable",
          foodId: cartItem.foodId,
          name: cartItem.name,
          message: `"${cartItem.name}" is no longer available`,
        });
        continue;
      }

      // 2) تغيّر السعر الأساسي (بدون حجم)
      if (!cartItem.size?.name) {
        if (food.price !== cartItem.basePrice) {
          changes.push({
            type: "price_changed",
            foodId: cartItem.foodId,
            name: cartItem.name,
            oldPrice: cartItem.basePrice,
            newPrice: food.price,
            message: `"${cartItem.name}" price changed from ${cartItem.basePrice} to ${food.price}`,
          });
        }
      }

      // 3) تغيّر سعر الحجم المختار
      if (cartItem.size?.name) {
        const freshSize = food.sizes?.find((s) => s.name === cartItem.size.name);
        if (!freshSize) {
          changes.push({
            type: "size_unavailable",
            foodId: cartItem.foodId,
            name: cartItem.name,
            size: cartItem.size.name,
            message: `Size "${cartItem.size.name}" for "${cartItem.name}" is no longer available`,
          });
        } else if (freshSize.price !== cartItem.size.price) {
          changes.push({
            type: "price_changed",
            foodId: cartItem.foodId,
            name: cartItem.name,
            size: cartItem.size.name,
            oldPrice: cartItem.size.price,
            newPrice: freshSize.price,
            message: `"${cartItem.name}" (${cartItem.size.name}) price changed from ${cartItem.size.price} to ${freshSize.price}`,
          });
        }
      }

      // 4) تغيّرت أسعار الإضافات
      if (cartItem.extras?.length > 0) {
        for (const cartExtra of cartItem.extras) {
          const freshExtra = food.extras?.find((e) => e.name === cartExtra.name);
          if (!freshExtra) {
            changes.push({
              type: "extra_unavailable",
              foodId: cartItem.foodId,
              name: cartItem.name,
              extra: cartExtra.name,
              message: `Extra "${cartExtra.name}" for "${cartItem.name}" is no longer available`,
            });
          } else if (freshExtra.price !== cartExtra.price) {
            changes.push({
              type: "price_changed",
              foodId: cartItem.foodId,
              name: cartItem.name,
              extra: cartExtra.name,
              oldPrice: cartExtra.price,
              newPrice: freshExtra.price,
              message: `Extra "${cartExtra.name}" for "${cartItem.name}" price changed from ${cartExtra.price} to ${freshExtra.price}`,
            });
          }
        }
      }
    }

    // ── إذا فيه تغييرات ولم يؤكد المستخدم بعد → أرجع التغييرات ──
    if (changes.length > 0 && !confirmed) {
      await session.abortTransaction();
      session.endSession();
      return res.status(200).json({
        requiresReview: true,
        changes,
        message: "Some items in your cart have changed. Please review before confirming.",
      });
    }

    // ── إذا فيه عناصر غير متاحة حتى بعد التأكيد → ارفض دائماً ──
    const hasUnavailable = changes.some(
      (c) => c.type === "unavailable" || c.type === "size_unavailable",
    );
    if (hasUnavailable) {
      await session.abortTransaction();
      session.endSession();
      return res.status(400).json({
        requiresReview: true,
        changes,
        message: "Some items are no longer available. Please remove them from your cart.",
      });
    }

    // ── بناء الـ items بالأسعار الحالية من DB ─────────────────
    let itemsPrice = 0;
    const items = cart.items.map((cartItem) => {
      const food = foodMap.get(cartItem.foodId.toString());

      // السعر الأساسي من DB
      let currentBasePrice = food.price;
      let currentSize = undefined;

      if (cartItem.size?.name) {
        const freshSize = food.sizes.find((s) => s.name === cartItem.size.name);
        currentBasePrice = freshSize.price;
        currentSize = { name: freshSize.name, price: freshSize.price };
      }

      // الإضافات بأسعارها الحالية من DB
      const currentExtras = cartItem.extras.map((cartExtra) => {
        const freshExtra = food.extras.find((e) => e.name === cartExtra.name);
        return { name: cartExtra.name, price: freshExtra ? freshExtra.price : cartExtra.price };
      });

      const extrasTotal = currentExtras.reduce((sum, e) => sum + e.price, 0);
      const totalPrice = parseFloat(((currentBasePrice + extrasTotal) * cartItem.quantity).toFixed(2));
      itemsPrice = parseFloat((itemsPrice + totalPrice).toFixed(2));

      return {
        foodId: cartItem.foodId,
        name: food.name,
        image: cartItem.image,
        price: currentBasePrice,
        size: currentSize,
        quantity: cartItem.quantity,
        extras: currentExtras,
        totalPrice,
      };
    });

    // ── حساب الإجماليات ────────────────────────────────────────
    const deliveryFee = restaurant.country === "DE" ? 3 : 1000;
    const taxRate = restaurant.taxRate || 0;

    const paymentMethod =
      restaurant.country === "DE"
        ? "card"
        : reqPaymentMethod || "cash";

    if (
      restaurant.country === "SY" &&
      !restaurant.paymentMethods.includes(paymentMethod)
    ) {
      await session.abortTransaction();
      session.endSession();
      return res.status(400).json({
        message: `Payment method "${paymentMethod}" is not accepted by this restaurant`,
      });
    }

    const foodTax = parseFloat(((itemsPrice * taxRate) / 100).toFixed(2));
    const deliveryTax =
      restaurant.country === "DE"
        ? parseFloat(((deliveryFee * 19) / 100).toFixed(2))
        : 0;
    const taxPrice = parseFloat((foodTax + deliveryTax).toFixed(2));
    const totalPrice = Number((itemsPrice + deliveryFee + taxPrice).toFixed(2));

    const orderNumber =
      "ORD-" + Date.now() + "-" + crypto.randomBytes(4).toString("hex");

    const order = await Order.create(
      [
        {
          userId,
          restaurantId: cart.restaurantId,
          orderNumber,
          items,
          itemsPrice,
          deliveryFee,
          taxPrice,
          totalPrice,
          deliveryAddress,
          paymentMethod,
          paymentStatus: "pending",
          notes,
        },
      ],
      { session },
    );

    await session.commitTransaction();
    session.endSession();

    // DE — نُرفق taxBreakdown في الـ response للعرض في الفاتورة
    const isGerman = restaurant.country === "DE";
    const responseOrder = order[0].toObject();
    if (isGerman) {
      responseOrder.taxBreakdown = {
        foodTax,
        foodTaxRate: taxRate,
        deliveryTax,
        deliveryTaxRate: 19,
        totalTax: taxPrice,
      };
    }

    res.status(201).json({
      requiresReview: false,
      message: "Order created successfully",
      order: responseOrder,
    });
  } catch (error) {
    await session.abortTransaction();
    session.endSession();
    res.status(500).json({ message: "Server error", error: error.message });
  }
};

⚠️ تغييرات في سلتك

// new DE — عند إنشاء الطلب، نحسب الضريبة بدقة ونُرفق تفصيلها في الـ response للعرض في الفاتورة
// payment
exports.createPaymentIntent = async (req, res) => {
  try {
    const userId = req.user._id ?? req.user.id;

    // الدفع الإلكتروني فقط للمستخدمين الألمان
    if (req.user.country !== "DE") {
      return res
        .status(400)
        .json({ message: "Payment intent is only for German users" });
    }

    const cart = await Cart.findOne({ userId });
    if (!cart || cart.items.length === 0) {
      return res.status(400).json({ message: "Cart is empty" });
    }

    const restaurant = await Restaurant.findById(cart.restaurantId).select(
      "taxRate country status",
    );
    if (!restaurant) {
      return res.status(404).json({ message: "Restaurant not found" });
    }
    if (restaurant.status !== "open") {
      return res.status(400).json({ message: "Restaurant is currently closed" });
    }

    // ── حساب المبلغ من DB مباشرة لا من السلة ─────────────────
    const foodIds = cart.items.map((i) => i.foodId);
    const freshFoods = await Food.find({ _id: { $in: foodIds } }).select(
      "price sizes extras status name",
    );
    const foodMap = new Map(freshFoods.map((f) => [f._id.toString(), f]));

    // تحقق من توفر العناصر قبل إنشاء الـ intent
    for (const cartItem of cart.items) {
      const food = foodMap.get(cartItem.foodId.toString());
      if (!food || food.status !== "available") {
        return res.status(400).json({
          message: `"${cartItem.name}" is no longer available. Please update your cart.`,
        });
      }
      if (cartItem.size?.name) {
        const freshSize = food.sizes?.find((s) => s.name === cartItem.size.name);
        if (!freshSize) {
          return res.status(400).json({
            message: `Size "${cartItem.size.name}" for "${cartItem.name}" is no longer available. Please update your cart.`,
          });
        }
      }
    }

    // حساب itemsPrice من DB
    let itemsPrice = 0;
    for (const cartItem of cart.items) {
      const food = foodMap.get(cartItem.foodId.toString());

      let currentBasePrice = food.price;
      if (cartItem.size?.name) {
        const freshSize = food.sizes.find((s) => s.name === cartItem.size.name);
        currentBasePrice = freshSize.price;
      }

      const currentExtras = cartItem.extras.map((cartExtra) => {
        const freshExtra = food.extras?.find((e) => e.name === cartExtra.name);
        return freshExtra ? freshExtra.price : cartExtra.price;
      });

      const extrasTotal = currentExtras.reduce((sum, p) => sum + p, 0);
      const itemTotal = parseFloat(
        ((currentBasePrice + extrasTotal) * cartItem.quantity).toFixed(2),
      );
      itemsPrice = parseFloat((itemsPrice + itemTotal).toFixed(2));
    }

    const deliveryFee = 3; // DE فقط
    const taxRate = restaurant.taxRate || 7;

    // DE: taxPrice = ضريبة الطعام (7%) + ضريبة التوصيل (19% — خدمة لوجستية)
    const foodTax = parseFloat(((itemsPrice * taxRate) / 100).toFixed(2));
    const deliveryTax = parseFloat(((deliveryFee * 19) / 100).toFixed(2));
    const taxPrice = parseFloat((foodTax + deliveryTax).toFixed(2));
    const totalPrice = Number((itemsPrice + deliveryFee + taxPrice).toFixed(2));

    // Stripe يعمل بالسنت كعملة صغيرة (cents)
    const amountInCents = Math.round(totalPrice * 100);

    const paymentIntent = await stripe.paymentIntents.create({
      amount: amountInCents,
      currency: "eur",
      metadata: {
        userId: userId.toString(),
        restaurantId: cart.restaurantId.toString(),
      },
    });

    res.status(200).json({
      success: true,
      clientSecret: paymentIntent.client_secret,
      paymentIntentId: paymentIntent.id,
      amount: totalPrice,
      currency: "EUR",
      // DE — تفصيل الضريبة للعرض في الفاتورة
      taxBreakdown: {
        foodTax,
        foodTaxRate: taxRate,
        deliveryTax,
        deliveryTaxRate: 19,
        totalTax: taxPrice,
      },
    });
  } catch (err) {
    console.error(err);
    res.status(500).json({ message: err.message });
  }
};
